```ts [nuxt.config.ts]
export default defineNuxtConfig({
  compatibilityDate: '2026-11-01',          // fixa o comportamento de presets do Nitro
  future: { compatibilityVersion: 5 },       // opcional: ativa as mudanças do Nuxt 5 desde já
  modules: ['@pinia/nuxt', '@nuxt/image', '@nuxt/eslint'],
  ssr: true,
  devtools: { enabled: true },

  app: {
    head: { htmlAttrs: { lang: 'pt-BR' }, titleTemplate: '%s · Loja MX' },
    pageTransition: { name: 'pagina', mode: 'out-in' },
  },
  css: ['~/assets/css/main.css'],

  runtimeConfig: { segredo: '', public: { apiBase: '/api' } },
  routeRules: { '/': { prerender: true } },

  $production: { routeRules: { '/**': { headers: { 'x-frame-options': 'DENY' } } } },
  $development: { devtools: { enabled: true } },

  nitro: { preset: 'node-server', storage: { cache: { driver: 'redis' } } },
  vite: { optimizeDeps: { include: ['lodash-es'] } },
  experimental: { ssrStreaming: false, payloadExtraction: true },
  extends: ['../base-layer'],               // layers: herdar config, pages, components de outro projeto
});
```

- **Layers** (`extends`) permitem compartilhar um "Nuxt base" entre projetos: componentes, composables, config e até páginas, sobrescrevíveis no projeto final.
- **`$production` / `$development` / `$env`** mesclam configurações por ambiente.
- **Hooks de build**: `hooks: { 'pages:extend'(pages) { … } }` altera rotas e outras etapas do build.
