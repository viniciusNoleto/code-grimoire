```js [main.js]
import { createApp } from 'vue';
import { createPinia } from 'pinia';
import App from './App.vue';
import { router } from './router';

const app = createApp(App);

app.use(createPinia());                        // plugins: objeto com install(app, opcoes)
app.use(router);
app.provide('config', { apiUrl: import.meta.env.VITE_API_URL }); // injeção para o app todo
app.component('BaseBotao', BaseBotao);         // componente global (evite em excesso: atrapalha tree-shaking)
app.directive('foco', vFoco);

app.config.errorHandler = (erro, instancia, info) => reportar(erro, info);
app.config.warnHandler = (msg) => { /* só dev */ };
app.config.globalProperties.$formatar = formatar; // acessível em templates; prefira provide/composables
app.config.idPrefix = 'app1';                  // para useId com vários apps na página

app.mount('#app');
// app.unmount() desmonta; createSSRApp para hidratar HTML vindo do servidor
```

### defineAsyncComponent e hidratação preguiçosa :version[3.5]

```js [async-component.js]
import { defineAsyncComponent, hydrateOnVisible, hydrateOnIdle } from 'vue';

const Grafico = defineAsyncComponent(() => import('./Grafico.vue'));   // code splitting

const Mapa = defineAsyncComponent({
  loader: () => import('./Mapa.vue'),
  loadingComponent: Carregando,
  errorComponent: FalhaAoCarregar,
  delay: 200,                     // só mostra o loading se demorar mais que isso
  timeout: 10000,
  hydrate: hydrateOnVisible(),    // SSR: só hidrata quando aparece na tela
});
// Também: hydrateOnIdle(), hydrateOnMediaQuery('(max-width: 600px)'), hydrateOnInteraction('click')
```

Hidratar menos de uma vez reduz long tasks e melhora o INP. O que `import()` faz no navegador e as prioridades de download estão em [Carregamento de recursos](/web#carregamento-de-recursos) e [Core Web Vitals](/web#core-web-vitals), no tomo Web.

### defineComponent, h e defineCustomElement

- **`defineComponent({ … })`**: não muda nada em runtime; serve para inferência de tipos fora de SFC com script setup.
- **`h(tipo, props, filhos)`**: cria VNodes em render functions. Útil para componentes muito dinâmicos e bibliotecas. A bancada "Quem renderiza?" do capítulo [03](#o-que-causa-re-render) é escrita assim.
- **`defineCustomElement(Componente)`**: transforma um componente Vue num Web Component nativo (`customElements.define('meu-botao', …)`) para usar em qualquer página. No sentido contrário, `app.config.compilerOptions.isCustomElement = tag => tag.includes('-')` diz ao compilador para não tratar tags com hífen como componentes Vue. Custom Elements, Shadow DOM e slots estão em [Web Components](/web#web-components), no tomo Web.
- **`data-allow-mismatch`** :version[3.5]: atributo que silencia avisos de hidratação esperados (datas, horas), como `suppressHydrationWarning` no React.
